### 计数循环含端点与步长
int s = 0
for i = 1 to 5
    s += i
print(s)
int t = 0
for i = 0 to 10 step 5
    t += i
print(t)
### repeat 与 continue
int n = 0
repeat(5)
    if (n == 2)
        continue
    n++
print(n)
### forever 加 break
int k = 0
forever
    k++
    if (k >= 3)
        break
print(k)
### try catch finally
fn f()
    try
        throw "坏了"
    catch e
        return "接住:" + e
    finally
        print("finally 跑了")

print(f())
### 类字段与方法
class P
    str name = "默认"
    int hp = 10

    fn hit(d)
        self.hp -= d
        return self.hp

P p = new P()
p.name = "小方"
print(p.name, p.hit(3))
### 类继承与 super init
class A
    int n = 0

    fn init(v)
        self.n = v

class B extends A
    fn init(v)
        super(v * 2)

B b = new B(3)
print(b.n)
