### 多行字典字面量
dict cfg = {
    "姓名": "阿方",
    "年龄": 18,
    "科目": ["语文", "数学"]
}
print(cfg["姓名"], len(cfg["科目"]))

### 多行列表字面量
list xs = [
    1,
    2,
    3
]
print(len(xs), xs[2])

### 嵌套容器跨行
dict 班 = {
    "学生": [
        {"名": "阿方", "分": 93},
        {"名": "小北", "分": 78}
    ],
    "名": "高三"
}
print(len(班["学生"]), 班["学生"][0]["分"])

### 函数调用参数跨行
print(
    "合计",
    1 + 2
)

### 续行中间夹注释
dict d = {
    /* 这是一行注释
    "a": 1,
    "b": 2
}
print(d["a"] + d["b"])

### 字符串里的括号不算深度
dict d = {
    "括号": "a)b(c",
    "n": 7
}
print(d["括号"], d["n"])

### 行尾块花括号仍然是块开始（回归）
int total = 0
for i in range(3) {
    total += i
}
if total == 3 {
    print("block ok", total)
}

### 单行字典不受影响（回归）
dict d = {"a": 1, "b": 2}
print(d["a"] + d["b"])
