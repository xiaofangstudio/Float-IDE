### 方法参数遮蔽字段：赋值只改局部
class P
    int hp = 10
    fn hit(hp)
        hp = 99
        return hp
p = new P()
print(p.hit(5))
print(p.hp)

### 方法里裸写字段名改的是字段
class Q
    int mp = 7
    fn cast()
        mp = 3
        return mp
q = new Q()
print(q.cast())
print(q.mp)
