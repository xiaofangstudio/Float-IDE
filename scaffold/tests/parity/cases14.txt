### f 拼接：字符串和数字混着写，不用强转
print(f{"总分：", 93, " 分"})

### f 拼接：不带引号的当变量
int n = 93
print(f{"总分：", n})

### f 拼接：项之间的空格忽略，字符串里的空格保留
print(f{  "a b"  ,  1  ,  "  c"  })

### f 拼接：全数字项不会被加成和
print(f{1, 2, 3})

### f 拼接：项可以是表达式
print(f{"1+2=", 1 + 2})

### f 拼接：列表和字典自动转显示文本
list xs = [1, 2]
dict d = {"名": "阿方"}
print(f{"列表 ", xs, " 字典 ", d})

### f 拼接：字符串里的逗号不当分隔符
print(f{"a,b", 1})

### f 拼接：参数里带逗号的函数调用不被切开
import data
print(f{"平均 ", data.avg([1, 2, 3])})

### f 拼接：空括号给空字符串
print("[" + f{} + "]")

### f 拼接：可以跨行写
print(f{
    "甲",
    1,
    "乙"
})

### f 拼接：f 仍然是普通变量名（回归）
f = 5
print(f)

### 行尾块花括号不受 f 影响（回归）
int total = 0
if 1 == 1 {
    total = f{1, 2}
}
print(total)

### 老的字符串插值不受影响（回归）
print("{1 + 2} 是 3")

### f 里面字符串仍支持插值
int n = 5
print(f{"n=", n, " 也写成 {n}"})
